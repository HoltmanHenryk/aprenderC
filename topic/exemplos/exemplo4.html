#include <stdio.h>

int main() {
    char cores[10][20];

    for (int i = 0; i < 5; i++) {
        printf("Digite sua cor favorita %d: ", i + 1);
        scanf("%s", cores[i]);
    }

    printf("Suas cores favoritas são:\n");
    for (int i = 0; i < 5; i++) {
        printf("%s\n", cores[i]);
    }

    return 0;
}
